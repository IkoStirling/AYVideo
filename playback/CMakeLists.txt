cmake_minimum_required(VERSION 3.20)
if(TARGET AYVideoPlayback)
    return()
endif()
get_filename_component(_video "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(_sources src/AYVideoBackendFactory.cpp src/AYVideoPlayer.cpp src/AYVideoSyncClock.cpp
    src/DecodeLoop.cpp src/VideoColorConvert.cpp src/AYVideoSubtitle.cpp
    backend/NullDemuxer.cpp backend/MockDemuxer.cpp backend/NullDecoder.cpp
    backend/MockDecoder.cpp backend/CpuVideoFrameTexture.cpp backend/MockVideoFrameTexture.cpp)
if(TARGET AYVideoDecode)
    set(_ffmpeg AYVideoDecode)
    get_target_property(_ffmpeg_dependency AYVideoDecode AYVIDEO_FFMPEG_DEPENDENCY)
elseif(NOT TARGET AYVideoDecode)
    list(APPEND _sources src/AYVideoTypes.cpp)
    find_package(FFMPEG QUIET)
    if(FFMPEG_FOUND)
        add_library(AYVideo_FFmpeg INTERFACE)
        target_include_directories(AYVideo_FFmpeg INTERFACE ${FFMPEG_INCLUDE_DIRS})
        target_link_directories(AYVideo_FFmpeg INTERFACE ${FFMPEG_LIBRARY_DIRS})
        target_link_libraries(AYVideo_FFmpeg INTERFACE ${FFMPEG_LIBRARIES})
        set(_ffmpeg AYVideo_FFmpeg)
        set(_ffmpeg_dependency AYVideo_FFmpeg)
        list(APPEND _sources backend/FFmpegDemuxer.cpp backend/FFmpegDecoder.cpp)
    endif()
endif()
if(_ffmpeg_dependency AND NOT TARGET AYVideo_FFmpeg)
    add_library(AYVideo_FFmpeg INTERFACE)
    target_link_libraries(AYVideo_FFmpeg INTERFACE ${_ffmpeg_dependency})
endif()
if(TARGET AYRenderer)
    list(APPEND _sources backend/RendererVideoFrameTexture.cpp)
endif()
list(TRANSFORM _sources PREPEND "${_video}/")
add_library(AYVideoPlayback STATIC ${_sources})
target_include_directories(AYVideoPlayback PUBLIC "${_video}" "${_video}/include" "${_video}/interface")
target_compile_features(AYVideoPlayback PUBLIC cxx_std_20)
target_compile_definitions(AYVideoPlayback PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(AYVideoPlayback PUBLIC AYTime AYAudioPlayback)
if(_ffmpeg)
    target_link_libraries(AYVideoPlayback PRIVATE ${_ffmpeg})
    target_compile_definitions(AYVideoPlayback PRIVATE AYVIDEO_HAS_FFMPEG=1)
endif()
if(TARGET AYRenderer)
    target_link_libraries(AYVideoPlayback PRIVATE AYRenderer)
    target_compile_definitions(AYVideoPlayback PRIVATE AYVIDEO_HAS_AYRENDERER=1)
endif()
if(BUILD_TESTING AND TARGET AYTest)
    file(GLOB _tests "${_video}/unittest/Test_*.cpp")
    list(FILTER _tests EXCLUDE REGEX "Test_(VideoSubSystem|VideoRuntimeModule|RendererFrameTexture)\\.cpp$")
    if(NOT _ffmpeg)
        list(FILTER _tests EXCLUDE REGEX "Test_(FFmpegDemuxer|FFmpegDecoder|PlayerPlayback|DecodeThread|AudioBridge)\\.cpp$")
    endif()
    add_executable(AYVideoPlayback_Tests "${_video}/unittest/main.cpp" ${_tests})
    target_link_libraries(AYVideoPlayback_Tests PRIVATE AYVideoPlayback AYTest)
    target_compile_definitions(AYVideoPlayback_Tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(_ffmpeg)
        target_sources(AYVideoPlayback_Tests PRIVATE "${_video}/unittest/FFmpegTestMedia.cpp")
        target_link_libraries(AYVideoPlayback_Tests PRIVATE AYVideo_FFmpeg)
    endif()
    ay_add_test(AYVideoPlayback_Tests)
    if(AY_VIDEO_DECODE_RUNTIME_FILES)
        add_custom_command(TARGET AYVideoPlayback_Tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${AY_VIDEO_DECODE_RUNTIME_FILES} "$<TARGET_FILE_DIR:AYVideoPlayback_Tests>" COMMAND_EXPAND_LISTS)
    endif()
    set_tests_properties(AYVideoPlayback_Tests PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
set(AYVIDEO_HAS_FFMPEG ${_ffmpeg} PARENT_SCOPE)
set(AYVIDEO_HAS_AYRENDERER FALSE PARENT_SCOPE)
if(TARGET AYRenderer)
    set(AYVIDEO_HAS_AYRENDERER TRUE PARENT_SCOPE)
endif()
