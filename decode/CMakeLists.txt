cmake_minimum_required(VERSION 3.25)
if(AY_VIDEO_DECODE_FFMPEG_TARGET AND TARGET ${AY_VIDEO_DECODE_FFMPEG_TARGET})
    set(_ay_decode_ffmpeg ${AY_VIDEO_DECODE_FFMPEG_TARGET})
else()
    find_package(FFMPEG REQUIRED)
    add_library(AYVideoDecode_FFmpeg INTERFACE)
    target_include_directories(AYVideoDecode_FFmpeg INTERFACE ${FFMPEG_INCLUDE_DIRS})
    target_link_directories(AYVideoDecode_FFmpeg INTERFACE ${FFMPEG_LIBRARY_DIRS})
    target_link_libraries(AYVideoDecode_FFmpeg INTERFACE ${FFMPEG_LIBRARIES})
    set(_ay_decode_ffmpeg AYVideoDecode_FFmpeg)
endif()
add_library(AYVideoDecode STATIC ../src/AYVideoDecodeSession.cpp ../src/AYVideoTypes.cpp ../backend/FFmpegDemuxer.cpp ../backend/FFmpegDecoder.cpp)
target_include_directories(AYVideoDecode PUBLIC ../include ../interface PRIVATE ../backend)
target_link_libraries(AYVideoDecode PUBLIC AYTime PRIVATE ${_ay_decode_ffmpeg})
set_property(TARGET AYVideoDecode PROPERTY AYVIDEO_FFMPEG_DEPENDENCY ${_ay_decode_ffmpeg})
target_compile_definitions(AYVideoDecode PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(AYVideoDecode PUBLIC cxx_std_20)
if(BUILD_TESTING AND TARGET AYTest)
    find_package(httplib CONFIG REQUIRED)
    add_executable(AYVideoDecode_Test Test_DecodeSession.cpp ../unittest/FFmpegTestMedia.cpp)
    target_link_libraries(AYVideoDecode_Test PRIVATE AYVideoDecode AYTest ${_ay_decode_ffmpeg} httplib::httplib)
    target_compile_definitions(AYVideoDecode_Test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    ay_add_test(AYVideoDecode_Test CASE_REPORT)
    if(AY_VIDEO_DECODE_RUNTIME_FILES)
        add_custom_command(TARGET AYVideoDecode_Test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${AY_VIDEO_DECODE_RUNTIME_FILES} "$<TARGET_FILE_DIR:AYVideoDecode_Test>" COMMAND_EXPAND_LISTS)
    endif()
endif()
